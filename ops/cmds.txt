======== Commands ========
kubectl -n testnet logs -l app=node --prefix --tail=20 --max-log-requests=63 # monitor all nodes
kubectl -n testnet logs deploy/node-0                                        # deploy node 0
kubectl -n testnet logs -l app=node --prefix --max-log-requests=63           # monitor all but node 0
kubectl -n testnet delete pod <name>                                         # kill a node
